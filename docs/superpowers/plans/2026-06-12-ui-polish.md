# UI Polish Implementation Plan

> **For agentic workers:** REQUIRED: Use superpowers:subagent-driven-development (if subagents available) or superpowers:executing-plans to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Apply five visual improvements: monitor-adaptive window startup size, longer splash with a smooth progress bar, status label moved inside the taskbar, and icon-only taskbar buttons drawn with ImDrawList primitives.

**Architecture:** All changes are additive edits to existing files — no new files, no CMakeLists changes. The `drawIcon` + `tbIconButton` helpers replace the existing `tbButton` helper in `taskbar.cpp`. The progress bar is appended to the existing `renderSplash` body. Window sizing is computed at startup from GLFW's primary monitor query.

**Tech Stack:** C++20, Dear ImGui v1.91.5 (ImDrawList API), GLFW 3.4, OpenGL 3.3, CMake 3.16+

**Spec:** `docs/superpowers/specs/2026-06-12-ui-polish-design.md`

- [ ] **Step 1: Add the `IconType` enum and `drawIcon` function**

  In `src/render/taskbar.cpp`, add `#include <cmath>` to the existing include block:

  Then, inside the `namespace csopesy {` block, **before** `renderTaskbar`, add (replacing the existing `tbButton` helper in Step 3 — for now, place this code right after the `namespace csopesy {` opening brace):

  ```cpp
  // Icon types for the taskbar icon-only buttons
  enum class IconType { Start, Files, Settings, TaskMgr, Vol, Net, Pwr };

  // Draws a minimal line-art icon centred on `c` within a `sz`-unit bounding box.
  // All strokes use 1.5 px thickness. ImDrawList has no round-cap support;
  // endpoints are flat (square). `sz` is the icon half-size (icon fits 2*sz × 2*sz).
  static void drawIcon(ImDrawList* dl, IconType type, ImVec2 c, float sz, ImU32 color) {
      const float lw = 1.5f;
      switch (type) {

      case IconType::Start: {
          // 2×2 grid of four stroked squares
          float s = sz * 0.40f;   // half-side of each cell
          float g = sz * 0.10f;   // half-gap between cells
          for (int row = 0; row < 2; ++row)
          for (int col = 0; col < 2; ++col) {
              float x = c.x + (col == 0 ? -(s * 2.0f + g) : g);
              float y = c.y + (row == 0 ? -(s * 2.0f + g) : g);
              dl->AddRect(ImVec2(x, y), ImVec2(x + s * 2.0f, y + s * 2.0f),
                          color, 0.0f, 0, lw);
          }
          break;
      }

      case IconType::Files: {
          // Folder body (rect) + small tab on top-left (three exposed sides)
          float hw = sz * 0.90f, hh = sz * 0.55f;  // body half-width/height
          float tw = sz * 0.42f, th = sz * 0.22f;  // tab width, height
          ImVec2 bMin(c.x - hw, c.y - hh + th);
          ImVec2 bMax(c.x + hw, c.y + hh);
          dl->AddRect(bMin, bMax, color, 0.0f, 0, lw);   // body
          // Tab: top, left, right edges (bottom edge == body top, no double-draw)
          ImVec2 tTL(c.x - hw,        c.y - hh);
          ImVec2 tTR(c.x - hw + tw,   c.y - hh);
          ImVec2 tBR(c.x - hw + tw,   c.y - hh + th);
          dl->AddLine(tTL, tTR, color, lw);   // tab top
          dl->AddLine(tTL, ImVec2(tTL.x, tBR.y), color, lw);  // tab left
          dl->AddLine(tTR, tBR, color, lw);   // tab right
          break;
      }

      case IconType::Settings: {
          // Center circle + 8 equidistant radial spokes
          float rCirc  = sz * 0.32f;
          float rInner = sz * 0.36f;
          float rOuter = sz * 0.52f;
          dl->AddCircle(c, rCirc, color, 0, lw);
          for (int i = 0; i < 8; ++i) {
              float a = i * (3.14159265f * 0.25f);
              dl->AddLine(
                  ImVec2(c.x + cosf(a) * rInner, c.y + sinf(a) * rInner),
                  ImVec2(c.x + cosf(a) * rOuter, c.y + sinf(a) * rOuter),
                  color, lw);
          }
          break;
      }

      case IconType::TaskMgr: {
          // Three vertical filled bars (bar chart), left-shortest to right-tallest
          const float bw  = sz * 0.22f;
          const float base = c.y + sz * 0.50f;
          const float xOff[3]    = { -sz * 0.36f, 0.0f, sz * 0.36f };
          const float heights[3] = {  sz * 0.42f, sz * 0.70f, sz * 0.92f };
          for (int i = 0; i < 3; ++i)
              dl->AddRectFilled(
                  ImVec2(c.x + xOff[i] - bw, base - heights[i]),
                  ImVec2(c.x + xOff[i] + bw, base), color);
          break;
      }

      case IconType::Vol: {
          // Speaker: rectangular body + two diagonal cone lines + closing front edge
          // + two diagonal lines for the sound arc
          float bLeft = c.x - sz * 0.46f, bRight = c.x - sz * 0.14f;
          float bTop  = c.y - sz * 0.28f, bBot   = c.y + sz * 0.28f;
          float cTipX = c.x + sz * 0.22f;
          float cTopY = c.y - sz * 0.48f, cBotY  = c.y + sz * 0.48f;
          // Body rect
          dl->AddRect(ImVec2(bLeft, bTop), ImVec2(bRight, bBot), color, 0.0f, 0, lw);
          // Cone (two diagonal lines from body right edge to the far tip)
          dl->AddLine(ImVec2(bRight, bTop), ImVec2(cTipX, cTopY), color, lw);
          dl->AddLine(ImVec2(bRight, bBot), ImVec2(cTipX, cBotY), color, lw);
          // Cone front edge
          dl->AddLine(ImVec2(cTipX, cTopY), ImVec2(cTipX, cBotY), color, lw);
          // Sound arc: two short diagonal lines
          float ax = c.x + sz * 0.30f;
          dl->AddLine(ImVec2(ax, c.y - sz * 0.20f), ImVec2(ax + sz * 0.16f, c.y - sz * 0.40f), color, lw);
          dl->AddLine(ImVec2(ax, c.y + sz * 0.20f), ImVec2(ax + sz * 0.16f, c.y + sz * 0.40f), color, lw);
          break;
      }

      case IconType::Net: {
          // Three vertical filled bars (signal strength), left-shortest to right-tallest
          const float bw   = sz * 0.22f;
          const float base = c.y + sz * 0.50f;
          const float xOff[3]    = { -sz * 0.36f, 0.0f, sz * 0.36f };
          const float heights[3] = {  sz * 0.32f, sz * 0.60f, sz * 0.92f };
          for (int i = 0; i < 3; ++i)
              dl->AddRectFilled(
                  ImVec2(c.x + xOff[i] - bw, base - heights[i]),
                  ImVec2(c.x + xOff[i] + bw, base), color);
          break;
      }

      case IconType::Pwr: {
          // Arc (circle with gap at top) + vertical stem through gap
          float r = sz * 0.44f;
          float gapA = 0.58f;  // radians — gap half-angle at the top of the circle
          // ImGui angle convention: 0 = right (+x), pi/2 = down (+y), -pi/2 = up (-y)
          float startA = -3.14159265f * 0.5f + gapA;  // just past the top-right
          float endA   =  3.14159265f * 1.5f - gapA;  // just before the top-left (= 3π/2 − gap)
          dl->PathArcTo(c, r, startA, endA, 32);
          dl->PathStroke(color, false, lw);
          // Vertical stem from icon top down to where it enters the arc gap
          dl->AddLine(ImVec2(c.x, c.y - sz * 0.52f),
                      ImVec2(c.x, c.y - r * 0.85f), color, lw);
          break;
      }

      } // switch
  }
  ```

- [ ] **Step 2: Add the `tbIconButton` helper**

  ```cpp
  // icon color default: cool blue-white; PWR overrides to reddish
  static bool tbIconButton(const char* id, IconType type, ImVec4 bg, float width,
                            ImU32 iconColor = IM_COL32(200, 215, 235, 220)) {
      ImVec2 pos  = ImGui::GetCursorScreenPos();
      ImVec2 size(width, 40.0f);
      bool clicked = ImGui::InvisibleButton(id, size);
      ImDrawList* dl = ImGui::GetWindowDrawList();
      dl->AddRectFilled(pos, ImVec2(pos.x + size.x, pos.y + size.y),
                        ImGui::ColorConvertFloat4ToU32(bg));
      drawIcon(dl, type,
               ImVec2(pos.x + size.x * 0.5f, pos.y + size.y * 0.5f),
               7.0f, iconColor);
      return clicked;
  }
  ```
